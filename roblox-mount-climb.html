1. LocalScript – Sistem Climbing
StarterCharacterScripts 
-- MountClimb LocalScript
-- Taruh di: StarterCharacterScripts > LocalScript
-- ============================================

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local UIS = game:GetService("UserInputService")

local player = Players.LocalPlayer
local char = player.Character or player.CharacterAdded:Wait()
local hum = char:WaitForChild("Humanoid")
local root = char:WaitForChild("HumanoidRootPart")

-- Konfigurasi
local CFG = {
    CLIMB_SPEED    = 12,
    MAX_STAMINA    = 100,
    STAMINA_DRAIN  = 0.8,
    STAMINA_REGEN  = 0.5,
    CHECK_DIST     = 2.8,
    WALL_OFFSET    = 2.5,
}

local stamina = CFG.MAX_STAMINA
local isClimbing = false
local climbPart = Instance.new("TrussPart")

-- Setup climb part (invisible)
climbPart.Size = Vector3.new(1, 6, 1)
climbPart.Transparency = 1
climbPart.CanCollide = false
climbPart.Anchored = false
climbPart.Parent = workspace

-- UI Stamina bar
local gui = Instance.new("ScreenGui", player.PlayerGui)
local frame = Instance.new("Frame", gui)
frame.Size = UDim2.new(0, 220, 0, 28)
frame.Position = UDim2.new(0.5, -110, 0, 12)
frame.BackgroundColor3 = Color3.fromRGB(20, 20, 30)

local bar = Instance.new("Frame", frame)
bar.Size = UDim2.new(1, 0, 1, 0)
bar.BackgroundColor3 = Color3.fromRGB(57, 255, 122)

-- Raycast deteksi dinding
local function checkWall()
    local dir = root.CFrame.LookVector * CFG.CHECK_DIST
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = {char, climbPart}
    params.FilterType = Enum.RaycastFilterType.Exclude
    return workspace:Raycast(root.Position, dir, params)
end

-- Main loop
RunService.Heartbeat:Connect(function(dt)
    local hit = checkWall()
    local climbKey = UIS:IsKeyDown(Enum.KeyCode.W)
        or UIS:IsKeyDown(Enum.KeyCode.Up)

    if hit and climbKey and stamina > 0 then
        isClimbing = true
        climbPart.CFrame = CFrame.new(hit.Position)
        hum:ChangeState(Enum.HumanoidStateType.Climbing)
        stamina = math.max(0, stamina - CFG.STAMINA_DRAIN)
    else
        isClimbing = false
        climbPart.CFrame = CFrame.new(0, -999, 0)
        stamina = math.min(CFG.MAX_STAMINA, stamina + CFG.STAMINA_REGEN)
    end

    bar.Size = UDim2.new(stamina / CFG.MAX_STAMINA, 0, 1, 0)

    if stamina == 0 and isClimbing then
        hum:ChangeState(Enum.HumanoidStateType.Freefall)
    end
end)
2. ServerScript – Game Manager
ServerScriptService 
-- MountClimb Server Script
-- Taruh di: ServerScriptService > Script
-- ======================================

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

-- RemoteEvent untuk sinkronisasi skor
local scoreEvent = Instance.new("RemoteEvent")
scoreEvent.Name = "ScoreUpdate"
scoreEvent.Parent = ReplicatedStorage

local scores = {}

-- Leaderboard setup
local function setupLeaderboard(player)
    local ls = Instance.new("Folder")
    ls.Name = "leaderstats"
    ls.Parent = player

    local height = Instance.new("IntValue")
    height.Name = "Ketinggian"
    height.Value = 0
    height.Parent = ls

    local score = Instance.new("IntValue")
    score.Name = "Skor"
    score.Value = 0
    score.Parent = ls

    scores[player.UserId] = {height = 0, score = 0}
end

-- Update skor dari client
scoreEvent.OnServerEvent:Connect(function(player, h, s)
    local ls = player:FindFirstChild("leaderstats")
    if ls then
        ls.Ketinggian.Value = math.max(ls.Ketinggian.Value, h)
        ls.Skor.Value = ls.Skor.Value + s
    end
end)

Players.PlayerAdded:Connect(setupLeaderboard)
Players.PlayerRemoving:Connect(function(p)
    scores[p.UserId] = nil
end)
3. ModuleScript – Mountain Generator
ReplicatedStorage 
-- MountainGenerator ModuleScript
-- Taruh di: ReplicatedStorage > ModuleScript
-- ==========================================

local MountainGen = {}

-- Generate gunung procedural
function MountainGen:Generate(config)
    config = config or {}
    local cfg = {
        baseX    = config.baseX    or 0,
        baseZ    = config.baseZ    or 0,
        height   = config.height   or 200,
        width    = config.width    or 80,
        steps    = config.steps    or 20,
        material = config.material or Enum.Material.Rock,
        color    = config.color    or Color3.fromRGB(120, 100, 80),
    }

    local folder = Instance.new("Folder")
    folder.Name = "Mountain"
    folder.Parent = workspace

    for i = 1, cfg.steps do
        local ratio = i / cfg.steps
        local w = cfg.width * (1 - ratio * 0.9)
        local y = (cfg.height / cfg.steps) * i
        local offset = math.random(-3, 3)

        local part = Instance.new("Part")
        part.Size = Vector3.new(w, 8, w)
        part.Position = Vector3.new(
            cfg.baseX + offset,
            y,
            cfg.baseZ + offset
        )
        part.Material = cfg.material
        part.BrickColor = BrickColor.new(cfg.color)
        part.Anchored = true
        part.Parent = folder

        -- Snow cap di puncak
        if i > cfg.steps * 0.8 then
            part.Material = Enum.Material.Snow
            part.BrickColor = BrickColor.new("White")
        end
    end

    return folder
end

return MountainGen
Panduan Instalasi di Roblox Studio
  • 1

    Buka Roblox Studio — buat experience baru pilih template Baseplate.

  • 2

    Import Script Climbing: Di Explorer, klik kanan StarterCharacterScripts → Insert Object → LocalScript. Paste isi script Sistem Climbing ke dalamnya.

  • 3

    Import Server Script: Klik kanan ServerScriptService → Insert Object → Script. Paste isi script Game Manager.

  • 4

    Tambah Mountain Generator: Klik kanan ReplicatedStorage → Insert Object → ModuleScript. Paste script Mountain Generator. Panggil dari script lain dengan require().

  • 5

    Buat Gunung: Di ServerScriptService tambah script baru, isi:
    local MG = require(game.ReplicatedStorage.ModuleScript)
    MG:Generate({height=300, width=100})

  • ✓

    Tekan F5 untuk Play Test. Karaktermu sudah bisa memanjat dinding gunung!

⚙ Kustomisasi: Ubah nilai CFG.CLIMB_SPEED, CFG.MAX_STAMINA, dan CFG.STAMINA_DRAIN di LocalScript untuk menyesuaikan kesulitan game.
